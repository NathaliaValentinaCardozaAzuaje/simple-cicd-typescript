document.getElementById('btnCalculate').addEventListener('click', function () { var a = Number(document.getElementById('num1').value) || 0; var b = Number(document.getElementById('num2').value) || 0; var op = document.getElementById('operation').value; var res; switch (op) { case '+': res = window.sum ? window.sum(a, b) : sum(a, b); break; case '-': res = window.substract(a, b); break; case '*': res = window.multiply(a, b); break; case '/': res = window.divide(a, b); break; } document.getElementById('result').textContent = res !== null ? res : 'Error'; });
